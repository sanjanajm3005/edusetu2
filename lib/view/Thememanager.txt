// theme_manager.dart
import 'package:flutter/material.dart';

class ThemeManager extends ChangeNotifier {
  static final ThemeManager _instance = ThemeManager._internal();
  factory ThemeManager() => _instance;
  ThemeManager._internal();

  ThemeData _currentTheme = lavenderTheme;
  ThemeData get currentTheme => _currentTheme;

  void setTheme(String themeName) {
    switch (themeName) {
      case "Mint Fresh":
        _currentTheme = mintTheme;
        break;
      case "Ocean Blue":
        _currentTheme = oceanTheme;
        break;
      case "Sunset Glow":
        _currentTheme = sunsetTheme;
        break;
      case "Lush Green":
        _currentTheme = greenTheme;
        break;
      default:
        _currentTheme = lavenderTheme;
    }
    notifyListeners();
  }
}

final ThemeData lavenderTheme = ThemeData(
  primaryColor: const Color(0xFFB589D6),
  scaffoldBackgroundColor: const Color(0xFFF3E9FB),
  colorScheme: const ColorScheme.light(
    primary: Color(0xFFB589D6),
    secondary: Color(0xFFDCC7F8),
  ),
);
